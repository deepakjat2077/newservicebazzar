# Everything under /user/ is account-facing — login/register/dashboard,
# and /user/profile/{username} too. That last one was briefly allowed
# (it does hold public review content), but that content already
# belongs to and shows on the listing page a review was left on; the
# profile URL itself is the user's own account page and should not show
# up in search on its own.
Disallow: /user/

# Search-result pages are query-driven and hold no unique content of their
# own — every result already lives on a canonical city/category page.
Disallow: /listings/search
Disallow: /search

# Internal AJAX/utility endpoints — JSON responses, not content pages, and
# some can't carry a noindex meta tag since they never render HTML.
Disallow: /ajax/
Disallow: /set-location-session
Disallow: /set-category-session

# Sorted and filtered views are the same listings in a different order, and
# the canonical tag on each already points back at the plain URL. Left
# crawlable they multiply every city/category page by every sort and filter
# combination, which is crawl budget spent to re-read pages already indexed.
#
# ?page= is deliberately NOT in this list: /blogs?page=2 and paginated
# listings pages carry a self-referencing canonical (see SiteController's
# blogs() and its listings pagination) — the code treats them as real,
# distinct, indexable pages. Blocking the crawl here made that canonical
# unreachable, so Google could never index page 2+ of anything paginated.
Disallow: /*?sort=
Disallow: /*?filter=
Disallow: /*?rating=
Disallow: /*?order=
Disallow: /*?location_id=

Allow: /
